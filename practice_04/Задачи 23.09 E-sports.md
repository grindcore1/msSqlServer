### Процедура 1. RegisterPlayer

**Код**

```sql
CREATE OR ALTER PROCEDURE dbo.RegisterPlayer
    @Nickname  NVARCHAR(50),
    @RealName  NVARCHAR(100),
    @Country   NVARCHAR(50),
    @BirthDate DATE,
    @TeamId    INT = NULL,
    @PlayerId  INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
    DECLARE @Msg NVARCHAR(400), @TeamName NVARCHAR(100), @IsActive BIT;
    BEGIN TRY
        IF DATEADD(YEAR, 16, @BirthDate) > CAST(GETDATE() AS DATE)
        BEGIN
            SET @Msg = CONCAT(N'Игроку ', @Nickname, N' меньше 16 лет (дата рождения ', CONVERT(NVARCHAR(10), @BirthDate, 104), N'). Регистрация невозможна.');
            THROW 50001, @Msg, 1;
        END;

        IF EXISTS (SELECT 1 FROM dbo.Players WHERE Nickname = @Nickname)
        BEGIN
            SET @Msg = CONCAT(N'Никнейм ', @Nickname, N' уже занят.');
            THROW 50002, @Msg, 1;
        END;

        IF @TeamId IS NOT NULL
        BEGIN
            SELECT @TeamName = Name, @IsActive = IsActive FROM dbo.Teams WHERE TeamId = @TeamId;
            IF @TeamName IS NULL
            BEGIN
                SET @Msg = CONCAT(N'Команда с TeamId = ', @TeamId, N' не существует.');
                THROW 50003, @Msg, 1;
            END;
            IF @IsActive = 0
            BEGIN
                SET @Msg = CONCAT(N'Команда ', @TeamName, N' неактивна. Регистрация в неё невозможна.');
                THROW 50004, @Msg, 1;
            END;
        END;

        BEGIN TRANSACTION;
            INSERT dbo.Players (Nickname, RealName, Country, BirthDate, TeamId)
            VALUES (@Nickname, @RealName, @Country, @BirthDate, @TeamId);
            SET @PlayerId = SCOPE_IDENTITY();
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;
```

**Тест**

```sql
DECLARE @NewId INT;

SELECT N'П1.1 Успешная регистрация' AS Test, @NewId AS NewPlayerId, Nickname, RealName, BirthDate, TeamId
FROM dbo.Players WHERE PlayerId = @NewId;
```

**Результат:** игрок Nova зарегистрирован без команды, NewPlayerId = 21.

### Процедура 2. CompleteMatch

**Код**

```sql
CREATE OR ALTER PROCEDURE dbo.CompleteMatch
    @MatchId    INT,
    @Team1Score INT,
    @Team2Score INT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
    DECLARE @Msg NVARCHAR(400), @Team1Id INT, @Team2Id INT, @IsCompleted BIT, @WinnerId INT;
    BEGIN TRY
        BEGIN TRANSACTION;
            SELECT @Team1Id = Team1Id, @Team2Id = Team2Id, @IsCompleted = IsCompleted
            FROM dbo.Matches WITH (UPDLOCK, HOLDLOCK)
            WHERE MatchId = @MatchId;

            IF @Team1Id IS NULL
            BEGIN
                SET @Msg = CONCAT(N'Матч с MatchId = ', @MatchId, N' не найден.');
                THROW 50101, @Msg, 1;
            END;
            IF @IsCompleted = 1
            BEGIN
                SET @Msg = CONCAT(N'Матч ', @MatchId, N' уже завершён. Повторно завершить нельзя.');
                THROW 50102, @Msg, 1;
            END;
            IF @Team1Score IS NULL OR @Team2Score IS NULL OR @Team1Score < 0 OR @Team2Score < 0
            BEGIN
                SET @Msg = CONCAT(N'Некорректный счёт матча ', @MatchId, N'.');
                THROW 50103, @Msg, 1;
            END;
            IF @Team1Score = @Team2Score
            BEGIN
                SET @Msg = CONCAT(N'Ничья недопустима (счёт ', @Team1Score, N':', @Team2Score, N').');
                THROW 50104, @Msg, 1;
            END;

            SET @WinnerId = CASE WHEN @Team1Score > @Team2Score THEN @Team1Id ELSE @Team2Id END;

            UPDATE dbo.Matches
            SET Team1Score = @Team1Score,
                Team2Score = @Team2Score,
                WinnerId = @WinnerId,
                IsCompleted = 1
            WHERE MatchId = @MatchId;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;
```

### Процедура 3. DistributePrizes

Финал определяется как матч последнего раунда, в котором встречаются победители предыдущего раунда, второй матч последнего раунда считается матчем за 3-е место.

**Код**

```sql
CREATE OR ALTER PROCEDURE dbo.DistributePrizes
    @TournamentId INT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
    DECLARE @Msg NVARCHAR(400), @Status NVARCHAR(20), @Pool DECIMAL(12,2), @MaxRound INT,
            @FinalId INT, @FinalDone BIT, @First INT, @Second INT, @Third INT;
    BEGIN TRY
        BEGIN TRANSACTION;
            SELECT @Status = Status, @Pool = PrizePool
            FROM dbo.Tournaments WITH (UPDLOCK, HOLDLOCK)
            WHERE TournamentId = @TournamentId;

            IF @Status IS NULL
            BEGIN
                SET @Msg = CONCAT(N'Турнир с TournamentId = ', @TournamentId, N' не найден.');
                THROW 50201, @Msg, 1;
            END;
            IF @Status <> N'Completed'
            BEGIN
                SET @Msg = CONCAT(N'Турнир ', @TournamentId, N' имеет статус ', @Status, N'. Распределять призы можно только для завершённого турнира.');
                THROW 50202, @Msg, 1;
            END;
            IF EXISTS (SELECT 1 FROM dbo.PrizeDistributions WHERE TournamentId = @TournamentId)
            BEGIN
                SET @Msg = CONCAT(N'Призы уже распределены для турнира ', @TournamentId, N'.');
                THROW 50203, @Msg, 1;
            END;

            SELECT @MaxRound = MAX(Round) FROM dbo.Matches WHERE TournamentId = @TournamentId;

            SELECT TOP (1) @FinalId = m.MatchId
            FROM dbo.Matches m
            WHERE m.TournamentId = @TournamentId AND m.Round = @MaxRound
            ORDER BY CASE WHEN EXISTS (SELECT 1 FROM dbo.Matches p
                                       WHERE p.TournamentId = @TournamentId AND p.Round = @MaxRound - 1 AND p.WinnerId = m.Team1Id)
                           AND EXISTS (SELECT 1 FROM dbo.Matches p
                                       WHERE p.TournamentId = @TournamentId AND p.Round = @MaxRound - 1 AND p.WinnerId = m.Team2Id)
                          THEN 0 ELSE 1 END,
                     m.MatchId;

            SELECT @FinalDone = IsCompleted,
                   @First = WinnerId,
                   @Second = CASE WHEN WinnerId = Team1Id THEN Team2Id ELSE Team1Id END
            FROM dbo.Matches
            WHERE MatchId = @FinalId;

            IF @FinalId IS NULL OR @FinalDone = 0 OR @First IS NULL
            BEGIN
                SET @Msg = CONCAT(N'Финальный матч турнира ', @TournamentId, N' не завершён.');
                THROW 50204, @Msg, 1;
            END;

            SELECT @Third = WinnerId
            FROM dbo.Matches
            WHERE TournamentId = @TournamentId AND Round = @MaxRound AND MatchId <> @FinalId AND IsCompleted = 1;

            INSERT dbo.PrizeDistributions (TournamentId, TeamId, Placement, Amount)
            SELECT @TournamentId, x.TeamId, x.Placement, ROUND(@Pool * x.Share, 2)
            FROM (VALUES (@First, 1, 0.50), (@Second, 2, 0.30), (@Third, 3, 0.20)) AS x (TeamId, Placement, Share)
            WHERE x.TeamId IS NOT NULL;

            UPDATE t
            SET t.TotalPrizeMoney = t.TotalPrizeMoney + pd.Amount
            FROM dbo.Teams t
            JOIN dbo.PrizeDistributions pd ON pd.TeamId = t.TeamId
            WHERE pd.TournamentId = @TournamentId;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;
```

### Функция 1. GetPlayerKDA

**Код**

```sql
CREATE OR ALTER FUNCTION dbo.GetPlayerKDA (@PlayerId INT)
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @KDA DECIMAL(10,2);
    SELECT @KDA = AVG(CAST(Kills + Assists AS DECIMAL(10,2)) / CASE WHEN Deaths = 0 THEN 1 ELSE Deaths END)
    FROM dbo.PlayerMatchStats
    WHERE PlayerId = @PlayerId;
    RETURN @KDA;
END;
```

    ORDER BY m.MatchDate DESC, m.MatchId DESC;

### Функция 3. GetTeamPowerRating (WITH SCHEMABINDING)

**Код**

```sql
CREATE OR ALTER FUNCTION dbo.GetTeamPowerRating (@TeamId INT)
RETURNS DECIMAL(12,2)
WITH SCHEMABINDING
AS
BEGIN
    DECLARE @Played INT, @Wins INT, @WinRate DECIMAL(9,4), @Prize DECIMAL(12,2), @Sponsor DECIMAL(12,2);

    SELECT @Played = COUNT(*),
           @Wins = SUM(CASE WHEN WinnerId = @TeamId THEN 1 ELSE 0 END)
    FROM dbo.Matches
    WHERE IsCompleted = 1 AND (Team1Id = @TeamId OR Team2Id = @TeamId);

    SET @WinRate = CASE WHEN @Played > 0 THEN CAST(@Wins AS DECIMAL(9,4)) / @Played ELSE 0 END;

    SELECT @Prize = TotalPrizeMoney FROM dbo.Teams WHERE TeamId = @TeamId;

    SELECT @Sponsor = ISNULL(SUM(ContributionAmount), 0) FROM dbo.TeamSponsors WHERE TeamId = @TeamId;

    RETURN @WinRate * 100 + ISNULL(@Prize, 0) / 10000 + @Sponsor / 5000;
END;
```

**Тест**

```sql
SELECT N'Ф3 Рейтинг силы' AS Test, t.TeamId, t.Name, dbo.GetTeamPowerRating(t.TeamId) AS PowerRating
FROM dbo.Teams t
WHERE t.IsActive = 1
ORDER BY PowerRating DESC;
```

**Результат:** Silent Storm 176.00 (100 + 52 + 24), Nexus Gaming 131.00 (100 + 15 + 16), Phoenix Reign 53.00, Void Hunters 13.50.

### Триггер 2. Автосоздание статистики при завершении матча

**Код**

```sql
CREATE OR ALTER TRIGGER dbo.trg_Matches_CreateStatsStubs
ON dbo.Matches
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    IF NOT UPDATE(IsCompleted) RETURN;

    INSERT dbo.PlayerMatchStats (PlayerId, MatchId, Kills, Deaths, Assists, IsMVP)
    SELECT p.PlayerId, i.MatchId, 0, 0, 0, 0
    FROM inserted i
    JOIN deleted d ON d.MatchId = i.MatchId
    JOIN dbo.Players p ON p.TeamId IN (i.Team1Id, i.Team2Id)
    WHERE d.IsCompleted = 0
      AND i.IsCompleted = 1
      AND NOT EXISTS (SELECT 1 FROM dbo.PlayerMatchStats s
                      WHERE s.PlayerId = p.PlayerId AND s.MatchId = i.MatchId);
END;
```

### Триггер 3. INSTEAD OF UPDATE на vw_PlayerTransfers

**Код**

```sql
CREATE OR ALTER VIEW dbo.vw_PlayerTransfers
AS
SELECT p.PlayerId,
       p.Nickname,
       p.RealName,
       p.Country AS PlayerCountry,
       p.TeamId,
       t.Name AS TeamName,
       t.IsActive AS TeamIsActive

LEFT JOIN dbo.Teams t ON t.TeamId = p.TeamId;
```

```sql
CREATE OR ALTER TRIGGER dbo.trg_vw_PlayerTransfers_Update
ON dbo.vw_PlayerTransfers
INSTEAD OF UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @Msg NVARCHAR(400);

    IF NOT UPDATE(TeamId) RETURN;

    IF EXISTS (SELECT 1 FROM inserted i
               WHERE NOT EXISTS (SELECT 1 FROM dbo.Teams t WHERE t.TeamId = i.TeamId AND t.IsActive = 1))
    BEGIN
        SELECT TOP (1) @Msg = CONCAT(N'Трансфер игрока ', i.Nickname, N' отменён: команда с TeamId = ',
                                     ISNULL(CAST(i.TeamId AS NVARCHAR(10)), N'NULL'), N' не существует или неактивна.')
        FROM inserted i
        WHERE NOT EXISTS (SELECT 1 FROM dbo.Teams t WHERE t.TeamId = i.TeamId AND t.IsActive = 1);
        ROLLBACK TRANSACTION;
        THROW 51301, @Msg, 1;
    END;

    IF EXISTS (SELECT 1 FROM inserted i JOIN dbo.Players p ON p.PlayerId = i.PlayerId WHERE p.TeamId = i.TeamId)
    BEGIN
        SELECT TOP (1) @Msg = CONCAT(N'Игрок ', p.Nickname, N' уже состоит в команде ', t.Name, N'.')
        FROM inserted i
        JOIN dbo.Players p ON p.PlayerId = i.PlayerId
        JOIN dbo.Teams t ON t.TeamId = p.TeamId
        WHERE p.TeamId = i.TeamId;
        ROLLBACK TRANSACTION;
        THROW 51302, @Msg, 1;
    END;

    IF EXISTS (SELECT 1
               FROM (SELECT TeamId, COUNT(*) AS Incoming FROM inserted GROUP BY TeamId) x
               WHERE (SELECT COUNT(*) FROM dbo.Players p
                      WHERE p.TeamId = x.TeamId
                        AND p.PlayerId NOT IN (SELECT PlayerId FROM inserted)) + x.Incoming > 5)
    BEGIN
        SELECT TOP (1) @Msg = CONCAT(N'Состав заполнен: в команде ', t.Name, N' уже 5 игроков.')
        FROM (SELECT TeamId, COUNT(*) AS Incoming FROM inserted GROUP BY TeamId) x
        JOIN dbo.Teams t ON t.TeamId = x.TeamId
        WHERE (SELECT COUNT(*) FROM dbo.Players p
               WHERE p.TeamId = x.TeamId
                 AND p.PlayerId NOT IN (SELECT PlayerId FROM inserted)) + x.Incoming > 5;
        ROLLBACK TRANSACTION;
        THROW 51303, @Msg, 1;
    END;

    UPDATE p
    SET p.TeamId = i.TeamId
    FROM dbo.Players p
    JOIN inserted i ON i.PlayerId = p.PlayerId;
END;
```

В тестовых данных в Void Hunters уже 5 игроков, поэтому трансфер Fade туда упал бы с ошибкой «Состав заполнен». Перед тестом из Void Hunters выведен игрок Dusk (TeamId = NULL), чтобы освободить место.

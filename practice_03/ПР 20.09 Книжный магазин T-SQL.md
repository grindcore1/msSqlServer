```sql
    AuthorId  INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Authors PRIMARY KEY,
    FullName  NVARCHAR(100) NOT NULL,
    Country   NVARCHAR(50)  NULL,
    BirthYear INT NULL CONSTRAINT CHK_Authors_BirthYear CHECK (BirthYear BETWEEN 1500 AND 2020)

    CategoryId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Categories PRIMARY KEY,
    Name       NVARCHAR(50) NOT NULL CONSTRAINT UQ_Categories_Name UNIQUE

CREATE TABLE dbo.Books (
    BookId      INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Books PRIMARY KEY,
    Title       NVARCHAR(200) NOT NULL,

    StockQty    INT NOT NULL CONSTRAINT DF_Books_StockQty DEFAULT 0
                CONSTRAINT CHK_Books_StockQty CHECK (StockQty >= 0),

    CONSTRAINT FK_Books_Authors FOREIGN KEY (AuthorId) REFERENCES dbo.Authors(AuthorId)
);

CREATE TABLE dbo.BookCategories (
    BookId     INT NOT NULL,
    CategoryId INT NOT NULL,
    CONSTRAINT PK_BookCategories PRIMARY KEY (BookId, CategoryId),
    CONSTRAINT FK_BookCategories_Books FOREIGN KEY (BookId) REFERENCES dbo.Books(BookId),
    CONSTRAINT FK_BookCategories_Categories FOREIGN KEY (CategoryId) REFERENCES dbo.Categories(CategoryId)
);

CREATE TABLE dbo.Customers (
    CustomerId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Customers PRIMARY KEY,
    FullName   NVARCHAR(100) NOT NULL,
    Email      NVARCHAR(100) NOT NULL CONSTRAINT UQ_Customers_Email UNIQUE,
    Phone      NVARCHAR(20)  NULL,
    City       NVARCHAR(50)  NOT NULL
);

CREATE TABLE dbo.Orders (
    OrderId    INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Orders PRIMARY KEY NONCLUSTERED,
    CustomerId INT NOT NULL,
    OrderDate  DATETIME2(0) NOT NULL CONSTRAINT DF_Orders_OrderDate DEFAULT SYSDATETIME(),
    Status     NVARCHAR(20) NOT NULL CONSTRAINT DF_Orders_Status DEFAULT N'Pending'
               CONSTRAINT CHK_Orders_Status CHECK (Status IN (N'Pending', N'Shipped', N'Delivered', N'Cancelled')),
    CONSTRAINT FK_Orders_Customers FOREIGN KEY (CustomerId) REFERENCES dbo.Customers(CustomerId)
);

CREATE TABLE dbo.OrderItems (
    OrderId   INT NOT NULL,
    BookId    INT NOT NULL,
    Quantity  INT NOT NULL CONSTRAINT CHK_OrderItems_Quantity CHECK (Quantity > 0),
    UnitPrice DECIMAL(10,2) NOT NULL CONSTRAINT CHK_OrderItems_UnitPrice CHECK (UnitPrice > 0),
    CONSTRAINT PK_OrderItems PRIMARY KEY (OrderId, BookId),
    CONSTRAINT FK_OrderItems_Orders FOREIGN KEY (OrderId) REFERENCES dbo.Orders(OrderId),
    CONSTRAINT FK_OrderItems_Books FOREIGN KEY (BookId) REFERENCES dbo.Books(BookId)
);
```

## Часть 3. Практические задания

### Задание 1. Индексы

**Код**

```sql
PRINT N'Задание 1';
CREATE CLUSTERED INDEX CIX_Orders_OrderDate ON dbo.Orders (OrderDate);
CREATE NONCLUSTERED INDEX IX_Books_AuthorId_Covering ON dbo.Books (AuthorId) INCLUDE (Title, Price, StockQty);
CREATE NONCLUSTERED INDEX IX_Orders_Pending ON dbo.Orders (OrderDate) INCLUDE (CustomerId) WHERE Status = N'Pending';
SELECT N'1 Индексы' AS Test, OBJECT_NAME(i.object_id) AS TableName, i.name AS IndexName,
       i.type_desc AS IndexType, i.has_filter AS HasFilter, i.filter_definition AS FilterDefinition
FROM sys.indexes i
WHERE i.name IN (N'CIX_Orders_OrderDate', N'IX_Books_AuthorId_Covering', N'IX_Orders_Pending');
```

**Результат:** созданы CIX_Orders_OrderDate (CLUSTERED), IX_Books_AuthorId_Covering (NONCLUSTERED), IX_Orders_Pending (NONCLUSTERED, фильтр ([Status]=N'Pending')).

- **CIX_Orders_OrderDate** - строки заказов физически упорядочены по дате, поэтому запросы за период (WHERE OrderDate BETWEEN ...) и сортировка по дате читают один непрерывный диапазон без сортировки.

- **IX_Books_AuthorId_Covering** - поиск книг автора идёт через Index Seek, а Title, Price, StockQty лежат в листьях индекса (INCLUDE), поэтому не нужен Key Lookup в таблицу.

- **IX_Orders_Pending** - индекс содержит только заказы со статусом Pending, он маленький и быстро отвечает на частый запрос «необработанные заказы», а при изменении других заказов не обновляется.

### Задание 7. MERGE

**Код**

```sql
PRINT N'Задание 7';
DROP TABLE IF EXISTS #BookStockStaging;
CREATE TABLE #BookStockStaging (
    BookId      INT NULL,
    NewStockQty INT NOT NULL,
    Title       NVARCHAR(200) NOT NULL,
    AuthorId    INT NOT NULL,
    Price       DECIMAL(10,2) NOT NULL
);
INSERT #BookStockStaging (BookId, NewStockQty, Title, AuthorId, Price) VALUES
(3, 20, N'Преступление и наказание', 2, 650.00),
(NULL, 10, N'Бесы', 2, 720.00);
MERGE dbo.Books AS t
USING #BookStockStaging AS s
    ON t.BookId = s.BookId
WHEN MATCHED THEN
    UPDATE SET t.StockQty = s.NewStockQty
WHEN NOT MATCHED BY TARGET THEN
    INSERT (Title, AuthorId, Price, StockQty) VALUES (s.Title, s.AuthorId, s.Price, s.NewStockQty)
OUTPUT N'7 MERGE' AS Test, $action AS MergeAction, inserted.BookId AS InsertedBookId, deleted.BookId AS DeletedBookId,
       deleted.StockQty AS OldStock, inserted.StockQty AS NewStock;
```

**Результат:** 2 действия: UPDATE для BookId 3 (остаток 0 -> 20) и INSERT новой книги «Бесы» с BookId 10 (deleted.BookId = NULL).

### Задание 10. Клиенты, купившие книги из 3+ категорий в одном заказе

**Код**

```sql
PRINT N'Задание 10';
SELECT N'10 3+ категорий' AS Test, cu.FullName AS Customer, o.OrderId,
       COUNT(DISTINCT bc.CategoryId) AS CategoriesCount
FROM dbo.Customers cu
JOIN dbo.Orders o ON o.CustomerId = cu.CustomerId
JOIN dbo.OrderItems oi ON oi.OrderId = o.OrderId
JOIN dbo.BookCategories bc ON bc.BookId = oi.BookId
GROUP BY cu.FullName, o.OrderId
HAVING COUNT(DISTINCT bc.CategoryId) >= 3
ORDER BY CategoriesCount DESC, o.OrderId;
```

**Результат:** 4 строки: Мария Сидорова - заказ 2 (5 категорий), Екатерина Волкова - заказ 7 (4), Иван Петров - заказ 1 (3), Мария Сидорова - заказ 8 (3).

Скриншот 16. Клиенты и заказы с 3 и более категориями

![[магазин20_16.png]]

Запрос работает потому, что схема нормализована: каждая книга заказа лежит отдельной строкой в OrderItems, а каждая категория книги - отдельной строкой в BookCategories. Благодаря этому можно соединить заказ с категориями через JOIN и посчитать COUNT(DISTINCT CategoryId) по заказу. Если бы категории хранились строкой через запятую, а книги заказа в одном поле, посчитать уникальные категории обычным JOIN было бы нельзя.

### Задание 11. Index Seek / Key Lookup

**Код**

```sql
DROP INDEX IF EXISTS IX_Books_AuthorId_Covering ON dbo.Books;
DROP INDEX IF EXISTS IX_Books_AuthorId ON dbo.Books;
GO
SELECT AuthorId, Title, Price FROM dbo.Books WHERE AuthorId = 1;
GO
CREATE NONCLUSTERED INDEX IX_Books_AuthorId ON dbo.Books (AuthorId);
GO
SELECT AuthorId, Title, Price FROM dbo.Books WITH (INDEX(IX_Books_AuthorId)) WHERE AuthorId = 1;
GO
DROP INDEX IX_Books_AuthorId ON dbo.Books;
CREATE NONCLUSTERED INDEX IX_Books_AuthorId_Covering ON dbo.Books (AuthorId) INCLUDE (Title, Price, StockQty);
GO
SELECT AuthorId, Title, Price FROM dbo.Books WHERE AuthorId = 1;
```

- **До покрывающего индекса:** индекса по AuthorId нет, поэтому будет Clustered Index Scan (при обычном некластерном индексе по AuthorId без INCLUDE - Index Seek + Key Lookup за Title и Price).

- **После покрывающего индекса:** только Index Seek по IX_Books_AuthorId_Covering, без Key Lookup, так как все нужные столбцы есть в индексе.
